\unexpanded\def\DefineComparisonTypeface#1#2{%
  \starttypescript[math][#1]
    \definefontsynonym[MathRoman]
      [file:\SomervilleFontDirectory/#2]
      [features={math\mathsizesuffix,mathextra,agate-comparison-axes}]
  \stoptypescript
  \definetypeface[#1][rm][serif][somerville-micro-semicondensed-semilight-text][default]%
  \definetypeface[#1][mm][math][#1][default]%
}

% Formula from somerville-letter-test.tex, lines 429-449.
\startbuffer[comparison-formula]
  \sum[size=26pt]{\frac{
    \bigcup{\frac{\bigcap{A_i}}{\bigsqcup{B_i}}}
  }{
    \bigoplus{\frac{\bigotimes{C_k}}{\bigsqcap{D_l}}}
  }}
  \prod[size=26pt]{\frac{
    \bigodot{\frac{\bigcup{E_i}}{\bigtimes{F_j}}}
  }{
    \coprod{\frac{\sum{a_k}}{\prod{b_l}}}
  }}
  \bigoplus[size=28pt]{\frac{
    \bigsqcup{\frac{\bigcup[size=4pt]{G_i}}{\bigcup{H_i}}}
  }{
    \bigotimes{\frac{\bigsqcap{K_k}}{\bigodot{L_l}}}
  }}\biguplus[size=26pt]{\frac{
    \bigsqcap{\frac{M_i}{N_j}}
  }{
    \sum{\frac{P_k}{Q_l}}
  }}\text{.}
\stopbuffer
