\unexpanded\def\LetterTestCell#1{%
  \bTD
    \switchtobodyfont[#1]%
    \getbuffer[lettertest]%
  \eTD
}

\unexpanded\def\LetterTestLabel#1{%
  \bTD
    \switchtobodyfont[somerville-regular,6pt]%
    #1%
  \eTD
}

\unexpanded\def\LetterTestOpticalTable#1#2{%
  \begingroup
  \def\testopticalsize{#1}%
  \def\testopticalcoordinate{#2}%
  \getbuffer[lettertest-table]%
  \blank
  \endgroup
}
