% Override Integral Slant in every named math instance in this test only.
% Preserve each instance's other axes and math goodies (including italic corrections).
\startluacode
for name, feature in table.sortedhash(fonts.specifiers.contextsetups) do
    if name:match("^somerville.*%-math%-axes$") and type(feature.axis) == "string" then
        local axes, count = feature.axis:gsub("[Ii][Nn][Ss][Ll]=[^,]+", "INSL=100")
        assert(count == 1, "Missing Integral Slant coordinate in " .. name)
        context.definefontfeature({ name }, { axis = axes, goodies = feature.goodies })
    end
end
\stopluacode

local references = structures.references
references.specials.lettertestview = function(var)
    local entries = references.collected[""] or { }
    local entry = entries[var.operation]
    local page = entry and entry.references.realpage
    if page then
        return lpdf.dictionary {
            S = lpdf.constant("GoTo"),
            D = lpdf.array {
                lpdf.reference(lpdf.pagereference(page)),
                lpdf.constant("XYZ"), lpdf.null(), lpdf.null(), lpdf.null(),
            },
        }
    end
end

% Derive navigation from the selected lists, including their endpoints.
\unexpanded\def\LetterTestRegisterNeighbor#1{%
  \setxvalue{LetterTestPrevious\testnavigationprefix:#1}{\testprevious}%
  \setvalue{LetterTestNext\testnavigationprefix:#1}{}%
  \ifx\testprevious\empty\else
    \setvalue{LetterTestNext\testnavigationprefix:\testprevious}{#1}%
  \fi
  \edef\testprevious{#1}%
}
\def\testnavigationprefix{Weight}
\let\testprevious\empty
\processcommacommand[\LetterTestWeights]\LetterTestRegisterNeighbor
\def\testnavigationprefix{Optical}
\let\testprevious\empty
\processcommacommand[\LetterTestOpticalSizes]\LetterTestRegisterNeighbor

% Draw arrows independently of the test font's glyph coverage.
\startuniqueMPgraphic{LetterTestUpArrow}
  drawarrow (0,0)--(0,5pt) withpen pencircle scaled .35pt;
  setbounds currentpicture to unitsquare xscaled 4pt yscaled 6pt
    shifted (-2pt,-.5pt);
\stopuniqueMPgraphic
\startuniqueMPgraphic{LetterTestDownArrow}
  drawarrow (0,5pt)--(0,0) withpen pencircle scaled .35pt;
  setbounds currentpicture to unitsquare xscaled 4pt yscaled 6pt
    shifted (-2pt,-.5pt);
\stopuniqueMPgraphic

\unexpanded\def\LetterTestNavigation#1#2#3{%
  \edef\testnext{\getvalue{LetterTestNext#1:#2}}%
  \edef\testprevious{\getvalue{LetterTestPrevious#1:#2}}%
  \ifx\testnext\empty\else
    \doifelse{#1}{Weight}
      {\edef\testdestination{lettertest:\testopticalsize:\testnext}}
      {\edef\testdestination{lettertest:\testnext:\testweight}}%
    \goto{\uniqueMPgraphic{LetterTestUpArrow}}[lettertestview(\testdestination)]\hskip.4em\relax%
  \fi
  #3%
  \ifx\testprevious\empty\else
    \doifelse{#1}{Weight}
      {\edef\testdestination{lettertest:\testopticalsize:\testprevious}}
      {\edef\testdestination{lettertest:\testprevious:\testweight}}%
    \hskip.4em\goto{\uniqueMPgraphic{LetterTestDownArrow}}[lettertestview(\testdestination)]%
  \fi
}

\unexpanded\def\LetterTestCell#1{%
  \bTD
    \switchtobodyfont[#1]%
    \getbuffer[lettertest]%
  \eTD
}

\unexpanded\def\LetterTestLabel#1{%
  \bTD
    \switchtobodyfont[somerville-regular,6pt]%
    #1%
  \eTD
}

\unexpanded\def\LetterTestWeightRow#1{%
  \lowercase{\def\testweightsuffix{-#1}}%
  \doif{#1}{Regular}{\let\testweightsuffix\empty}%
  \bTR
    \LetterTestLabel{\LetterTestNavigation{Weight}{#1}{#1}}%
    \expanded{\LetterTestCell{somerville-\testopticalsize-semicondensed\testweightsuffix}}%
    \expanded{\LetterTestCell{somerville-\testopticalsize\testweightsuffix}}%
    \expanded{\LetterTestCell{somerville-\testopticalsize-semiexpanded\testweightsuffix}}%
  \eTR
}

% One weight per page keeps each optical-size header's links tied to that weight.
\unexpanded\def\LetterTestWeightTable#1{%
  \def\testweight{#1}%
  \pagereference[lettertest:\testopticalsize:#1]%
  \getbuffer[lettertest-table]%
%  \page
}

\unexpanded\def\LetterTestOpticalTable#1#2{%
  \begingroup
  \def\testopticalsize{#1}%
  \def\testopticalcoordinate{#2}%
  \processcommacommand[\LetterTestWeights]\LetterTestWeightTable
  \endgroup
}

\unexpanded\def\LetterTestNamedOpticalTable#1{%
  \LetterTestOpticalTable{#1}{\getvalue{LetterTestOpticalCoordinate#1}}%
}
